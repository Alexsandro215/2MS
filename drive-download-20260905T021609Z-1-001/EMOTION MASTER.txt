package "EMOCIONES DISPONIBLES" {

enum EmocionTipo {
  FURIA
  ALEGRIA
  TRISTEZA
  MIEDO
  ENAMORADO
  ALTERADO
  SERIO
  ENFERMO
  NEUTRAL
}

class Furia {
  +bonusDanioFisico = +50% del total actual
  +restriccion = no elige target durante 1 turno
  +naturaleza = agresiva / impulsiva
}

class Alegria {
  +bonus = gana 1 turno extra al inicio de cada batalla
}

class Tristeza {
  +restriccionTurno1 = no puede atacar ni defender
  +bonusPasivasYGadgets = +10% a +40% segun nivel
  +baseDeEscalado = stats base afectadas en ese momento
}

class Miedo {
  +accionAutomaticaTurno1 = DEFENSA
  +bonus = la defensa no se rompe al primer impacto
  +duracionInterna = todo el turno 1
}

class Enamorado {
  +penalizacion = pierde el primer turno
  +snapshotInicioCombate = true
  +calculoOcultoAlInicio = true
  +seleccion = villano aleatorio
  +regla = toma la stat secundaria mas alta del villano seleccionado
  +bonus = obtiene 50% de esa categoria usando sus propias stats base
  +duracion = todo el combate
  +recalculo = no
}

class Alterado {
  +penalizacionPunteriaTurno1 = -50% del total actual
  +bonusActivacionElemento = +50% relativo sobre la probabilidad base actual
}

class Serio {
  +bonusCritico = +5% por cada 10% de vida base faltante
}

class Enfermo {
  +penalizacionStats = todas las stats a 2/3 durante los primeros 3 turnos
  +excepciones = vida y desgaste no cuentan
  +bonusCargaMusical = x1.5 sobre toda fuente de obtencion de CM
  +aplicacionCM = por cada fuente de forma independiente
}

EmocionTipo --> Furia
EmocionTipo --> Alegria
EmocionTipo --> Tristeza
EmocionTipo --> Miedo
EmocionTipo --> Enamorado
EmocionTipo --> Alterado
EmocionTipo --> Serio
EmocionTipo --> Enfermo
}

note right of SistemaEmocional
- El sistema maneja emociones ON-GAME y OFF-GAME
- Cada personaje posee reglas propias
- La emocion activa dura el dia / run
- Cada emocion define su propia ventana de aplicacion al entrar en combate
- No existe una potenciacion general compartida de 3 turnos
- Para cambiar de emocion primero debe limpiarse y pasar a NEUTRAL
- Al final del dia todas las emociones se eliminan
- Los modificadores se integran al Calculus Master
end note
